# SoftDelete

``````
pnpm psy sync

```

SoftDelete works with `deletedAt`, which current generators include by default. If adding after the fact, generate a migration with an optional `deletedAt` datetime:

```console
pnpm psy g:migration add-deleted-at-to-rooms deleted_at:datetime:optional
```

To cascade deletion (regular or soft), add `dependent: 'destroy'` to the relevant association declarations.

A soft-deleted row stays in its table, so a plain unique constraint on a natural key would keep reserving it. The unique constraints on `users.email`, `guests.user_id` and `hosts.user_id` are replaced with unique indexes that only constrain live rows (`WHERE deleted_at IS NULL`), so, for example, a new User can reuse the email address of a destroyed User:

```console
pnpm psy g:migration make-natural-key-unique-indexes-partial
pnpm psy db:migrate
```
``````

``````
+import { DreamMigrationHelpers } from '@rvoh/dream/db'
+import { Kysely, sql, type SqlBool } from 'kysely'
+
+// Users, Guests and Hosts are soft-deleted, which leaves their rows in the table.
+// A plain unique constraint would keep reserving a soft-deleted row's email or
+// user, so each is replaced by a unique index that only constrains live rows.
+const NATURAL_KEYS = [
+  { table: 'users', column: 'email', constraint: 'users_email_key', index: 'users_email_unique' },
+  { table: 'guests', column: 'user_id', constraint: 'guests_user_id_key', index: 'guests_user_id_unique' },
+  { table: 'hosts', column: 'user_id', constraint: 'hosts_user_id_key', index: 'hosts_user_id_unique' },
+] as const
+
+// eslint-disable-next-line @typescript-eslint/no-explicit-any
+export async function up(db: Kysely<any>): Promise<void> {
+  for (const { table, column, constraint, index } of NATURAL_KEYS) {
+    await DreamMigrationHelpers.dropConstraint(db, constraint, { table })
+
+    await db.schema
+      .createIndex(index)
+      .on(table)
+      .column(column)
+      .unique()
+      .where(sql<SqlBool>`deleted_at IS NULL`)
+      .execute()
+  }
+}
+
+// eslint-disable-next-line @typescript-eslint/no-explicit-any
+export async function down(db: Kysely<any>): Promise<void> {
+  for (const { table, column, constraint, index } of NATURAL_KEYS) {
+    await db.schema.dropIndex(index).execute()
+
+    await db.schema
+      .alterTable(table)
+      .addUniqueConstraint(constraint, [column])
+      .execute()
+  }
``````
